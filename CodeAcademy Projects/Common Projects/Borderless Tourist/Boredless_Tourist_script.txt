# 38-46. Define find_attractions(destination, interests)
def find_attractions(destination, interests):
    """Return a list of attraction *names* that match any of the given interests."""
    destination_index = get_destination_index(destination)
    attractions_in_city = attractions[destination_index]
    attractions_with_interest = []

    for possible_attraction in attractions_in_city:
        attraction_tags = possible_attraction[1]
        for interest in interests:
            if interest in attraction_tags:
                # Append only the name (task 49)
                attractions_with_interest.append(possible_attraction[0])
                break  # avoid adding the same attraction twice
    return attractions_with_interest

# 53-60. Define get_attractions_for_traveler(traveler)
def get_attractions_for_traveler(traveler):
    """Build and return a friendly recommendation message for the traveler."""
    traveler_destination = traveler[1]
    traveler_interests = traveler[2]
    traveler_attractions = find_attractions(traveler_destination, traveler_interests)

    interests_string = "Hi " + traveler[0]
    interests_string += ", we think you'll like these places around "
    interests_string += traveler_destination + ": "
    interests_string += ", ".join(traveler_attractions)
    return interests_string

# --- Alternate 1: list comprehension + any() ---
def find_attractions_alt(destination, interests):
    """Alternate implementation using list comprehension and any()."""
    destination_index = get_destination_index(destination)
    attractions_in_city = attractions[destination_index]
    return [
        name
        for name, tags in attractions_in_city
        if any(interest in tags for interest in interests)
    ]


print("\nAlternate find_attractions (LA art):", find_attractions_alt("Los Angeles, USA", ["art"]))


# --- Alternate 2: dictionary-based storage ---
attractions_dict = {city: [] for city in destinations}

def add_attraction_dict(destination, attraction):
    attractions_dict[destination].append(attraction)

def find_attractions_dict(destination, interests):
    return [
        name
        for name, tags in attractions_dict[destination]
        if any(interest in tags for interest in interests)
    ]

# Populate the dict from the existing list structure
for city, atts in zip(destinations, attractions):
    for att in atts:
        add_attraction_dict(city, att)

print("Dict version – Shanghai art/historical:",
      find_attractions_dict("Shanghai, China", ["art", "historical site"]))


# --- Alternate 3: cleaner f-string message ---
def get_attractions_for_traveler_f(traveler):
    name, dest, interests = traveler
    places = find_attractions(dest, interests)
    place_list = ", ".join(places) if places else "nothing we currently have tagged"
    return f"Hi {name}, we think you'll like these places around {dest}: {place_list}"

# --- Tag helper ---
def attractions_by_tag(tag):
    """Return {city: [attraction names]} for every attraction containing the tag."""
    result = {}
    for city, atts in zip(destinations, attractions):
        matched = [name for name, tags in atts if tag in tags]
        if matched:
            result[city] = matched
    return result
